import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    let fixtures = app.cells.containing(.staticText, identifier:"Audit Fixtures").firstMatch
    XCTAssertTrue(fixtures.waitForExistence(timeout:5))
    fixtures.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.25)).tap()
    let file = app.cells.containing(.staticText, identifier:"manuscript").firstMatch
    XCTAssertTrue(file.waitForExistence(timeout:5))
    file.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.25)).tap()
    let open = app.buttons["DOCPicker.actionButton"]
    open.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.5)).tap()
    XCTAssertTrue(open.waitForNonExistence(timeout:5))
    XCTAssertTrue(app.buttons["Import"].firstMatch.waitForExistence(timeout:5))

  }
}
