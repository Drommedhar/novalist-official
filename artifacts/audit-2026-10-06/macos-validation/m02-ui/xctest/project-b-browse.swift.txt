import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    let name = app.textFields["My Novel"]
    XCTAssertTrue(name.waitForExistence(timeout:5))
    name.tap()
    name.typeText("Audit Project B")
    app.buttons["Browse..."].firstMatch.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.5)).tap()
    let parent = app.cells.containing(.staticText, identifier:"FixtureParent").firstMatch
    XCTAssertTrue(parent.waitForExistence(timeout:5))
    parent.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.25)).tap()
    XCTAssertTrue(app.staticTexts["SiblingA"].firstMatch.waitForExistence(timeout:5))
    let open = app.buttons["DOCPicker.actionButton"]
    open.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.5)).tap()
    XCTAssertTrue(open.waitForNonExistence(timeout:5))

  }
}
