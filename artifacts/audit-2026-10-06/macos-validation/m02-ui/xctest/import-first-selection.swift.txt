import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    let choose = app.buttons["Choose a manuscript..."].firstMatch
    XCTAssertTrue(choose.waitForExistence(timeout:5))
    choose.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.5)).tap()
    let open = app.buttons["DOCPicker.actionButton"]
    XCTAssertTrue(open.waitForExistence(timeout:5))
    if !app.staticTexts["manuscript"].firstMatch.exists {
      app.buttons["DOC.navBarButton.backInHistory"].tap()
    }
    let cell = app.cells.containing(.staticText, identifier:"manuscript").firstMatch
    XCTAssertTrue(cell.waitForExistence(timeout:5))
    cell.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.25)).tap()
    XCTAssertTrue(open.isEnabled)
    open.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.5)).tap()
    XCTAssertTrue(open.waitForNonExistence(timeout:5))

  }
}
