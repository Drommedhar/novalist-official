import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    app.buttons["Browse..."].firstMatch.tap()
    XCTAssertTrue(app.buttons["DOCPicker.actionButton"].waitForExistence(timeout:5))
    app.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.075)).press(forDuration:0.1,thenDragTo:app.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.8)))
    XCTAssertTrue(app.buttons["Create Project"].waitForExistence(timeout:5))

  }
}
