import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    let card = app.descendants(matching:.any).matching(NSPredicate(format:"label == %@", "Select Audit Project B — Audit Project B")).firstMatch
    XCTAssertTrue(card.waitForExistence(timeout:8))
    card.coordinate(withNormalizedOffset:CGVector(dx:0.5,dy:0.5)).doubleTap()
    XCTAssertTrue(app.staticTexts["Project Dashboard"].firstMatch.waitForExistence(timeout:10))
    XCTAssertTrue(app.staticTexts["Audit Project B"].firstMatch.exists)
    XCTAssertFalse(app.buttons["DOCPicker.actionButton"].exists)

  }
}
