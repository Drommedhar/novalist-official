import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    let browse = app.buttons["Browse for Project Folder..."].firstMatch
    XCTAssertTrue(browse.waitForExistence(timeout: 10))
    browse.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).tap()
    XCTAssertTrue(app.buttons["DOCPicker.actionButton"].firstMatch.waitForExistence(timeout: 10))

  }
}
