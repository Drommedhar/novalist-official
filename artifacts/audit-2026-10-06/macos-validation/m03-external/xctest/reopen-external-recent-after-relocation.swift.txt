import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    let external = app.descendants(matching: .any).matching(NSPredicate(format: "label == %@", "Select Audit External Book — Audit External Match")).firstMatch
    XCTAssertTrue(external.waitForExistence(timeout: 12))
    XCTAssertFalse(app.buttons["DOCPicker.actionButton"].exists)
    external.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).doubleTap()
    XCTAssertTrue(app.staticTexts["Project Dashboard"].firstMatch.waitForExistence(timeout: 12))
    XCTAssertFalse(app.buttons["DOCPicker.actionButton"].exists)
    app.buttons["Write"].firstMatch.tap()
    XCTAssertTrue(app.buttons["Export"].firstMatch.waitForExistence(timeout: 10))
    app.coordinate(withNormalizedOffset: CGVector(dx: 0.30, dy: 0.389)).tap()
    XCTAssertTrue(app.textViews.firstMatch.waitForExistence(timeout: 10))
    XCTAssertTrue((app.textViews.firstMatch.value as? String ?? "").contains("M03EXTERNAL20261007"))

  }
}
