import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    let folder = app.cells.containing(.staticText, identifier: "MovedParent").firstMatch
    XCTAssertTrue(folder.waitForExistence(timeout: 10))
    folder.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.25)).tap()
    XCTAssertTrue(app.staticTexts["SiblingA"].firstMatch.waitForExistence(timeout: 10))
    app.buttons["Sichern"].firstMatch.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).tap()
    XCTAssertTrue(app.buttons["Sichern"].firstMatch.waitForNonExistence(timeout: 10))
    XCTAssertTrue(app.staticTexts["Export complete."].firstMatch.waitForExistence(timeout: 10))

  }
}
