import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    let title = app.textFields.matching(NSPredicate(format: "value == %@", "Audit Project A")).firstMatch
    XCTAssertTrue(title.waitForExistence(timeout: 10))
    XCTAssertTrue(app.buttons["Export"].firstMatch.isEnabled)
    app.buttons["Export"].firstMatch.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).tap()
    XCTAssertTrue(app.buttons["In Dateien sichern"].firstMatch.waitForExistence(timeout: 20))

  }
}
