import XCTest
final class AuditUITests: XCTestCase {
  func testM02NativePicker() throws {
    continueAfterFailure = false
    let app = XCUIApplication(bundleIdentifier: "com.novalist.app")
    app.activate()
    XCTAssertTrue(app.staticTexts["Export complete."].firstMatch.waitForExistence(timeout: 10))
    let start = Date()
    app.buttons["Export"].firstMatch.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).tap()
    let activity = app.descendants(matching: .any).matching(NSPredicate(format: "label == %@", "In Dateien sichern")).firstMatch
    XCTAssertTrue(activity.waitForExistence(timeout: 10))
    activity.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).tap()
    let save = app.buttons["Sichern"].firstMatch
    XCTAssertTrue(save.waitForExistence(timeout: 10))
    let fixture = app.cells.containing(.staticText, identifier: "Audit Fixtures").firstMatch
    if fixture.exists {
      fixture.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.25)).tap()
    }
    XCTAssertTrue(app.staticTexts["MovedParent"].firstMatch.waitForExistence(timeout: 10))
    save.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).tap()
    let replace = app.buttons["Ersetzen"].firstMatch
    if replace.waitForExistence(timeout: 3) {
      replace.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).tap()
    }
    XCTAssertTrue(save.waitForNonExistence(timeout: 10))
    XCTAssertTrue(app.staticTexts["Export complete."].firstMatch.waitForExistence(timeout: 10))
    print("QUICK_SHARE_COMPLETED_SECONDS " + String(Date().timeIntervalSince(start)))

  }
}
